cmake_minimum_required(VERSION 3.24)
# IDE metadata only; isolated owner execution lives in tests/sesh/run.py.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(sesh_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Local Vexspoke src; no downloads")
set(API_HAVEN_SOURCE_DIR "" CACHE PATH "Local API Haven src; no downloads")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "src/*.c" "src/*.h")
add_library(sesh_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(sesh_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
foreach(dependency VEXSPOKE_SOURCE_DIR API_HAVEN_SOURCE_DIR)
    if(${dependency})
        target_include_directories(sesh_ide PRIVATE "${${dependency}}")
    endif()
endforeach()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(sesh_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(sesh_ide PRIVATE -mcpu=apple-m1)
    endif()
endif()
message(STATUS "IDE metadata only; supply local dependency roots. Integrated builds belong to b.")
